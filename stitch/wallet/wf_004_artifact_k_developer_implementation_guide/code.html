async function commitFunding(txId) {
  try {
    const response = await mTLS.post('/ledger/commit', {
      transaction_id: txId,
      integrity_hash: hmac(txId, SECRET)
    });

    if (response.code === 200) {
      return state.transition('SUCCESS');
    }
  } catch (err) {
    return state.compensate('ROLLBACK_FUNDING');
  }
